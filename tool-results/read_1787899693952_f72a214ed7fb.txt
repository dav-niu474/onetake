   144→/** 自动分层布局 */
   145→function autoLayout() {
   146→  const { nodes, edges } = useCanvasStore.getState()
   147→  if (nodes.length === 0) return
   148→  const depth: Record<string, number> = {}
   149→  nodes.forEach((n) => (depth[n.id] = 0))
   150→  for (let i = 0; i < nodes.length; i++) {
   151→    edges.forEach((e) => {
   152→      depth[e.target] = Math.max(depth[e.target] ?? 0, (depth[e.source] ?? 0) + 1)
   153→    })
   154→  }
   155→  const layers = new Map<number, string[]>()
   156→  nodes.forEach((n) => {
   157→    const d = depth[n.id] ?? 0
   158→    layers.set(d, [...(layers.get(d) ?? []), n.id])
   159→  })
   160→  const pos: Record<string, { x: number; y: number }> = {}
   161→  layers.forEach((ids, d) => {
   162→    ids.forEach((id, i) => {
   163→      pos[id] = { x: d * 400, y: i * 280 + 60 }
   164→    })
   165→  })
   166→  useCanvasStore.setState({
   167→    nodes: nodes.map((n) => ({ ...n, position: pos[n.id] ?? n.position })),
   168→    dirty: true,
   169→  })
   170→}

   410→  /* ---------- 拖拽对齐参考线（Smart Alignment Guides） ---------- */
   411→  /**
   412→   * 拖动节点时与其他节点的左/中/右、上/中/下边缘比对：
   413→   * 屏幕距离阈值 6px 内吸附到精确对齐，并显示虚线参考线。
   414→   * 多选拖拽 / 隐藏节点（折叠分组成员）不参与。
   415→   */
   416→  const ALIGN_THRESHOLD_SCREEN = 6
   417→  const applyAlignment = useCallback(
   418→    (dragged: Node<CanvasNodeData>, all: Node<CanvasNodeData>[]) => {
   419→      const zoom = getZoom()
   420→      const threshold = ALIGN_THRESHOLD_SCREEN / zoom
   421→      const dw = dragged.measured?.width ?? 0
   422→      const dh = dragged.measured?.height ?? 0
   423→      const dxs = [dragged.position.x, dragged.position.x + dw / 2, dragged.position.x + dw]
   424→      const dys = [dragged.position.y, dragged.position.y + dh / 2, dragged.position.y + dh]
   425→
   426→      let bestV: { diff: number; line: number } | null = null
   427→      let bestH: { diff: number; line: number } | null = null
   428→
   429→      for (const other of all) {
   430→        if (other.id === dragged.id || other.hidden) continue
   431→        const ow = other.measured?.width ?? 0
   432→        const oh = other.measured?.height ?? 0
   433→        const oxs = [other.position.x, other.position.x + ow / 2, other.position.x + ow]
   434→        const oys = [other.position.y, other.position.y + oh / 2, other.position.y + oh]
   435→        for (const dx of dxs) {
   436→          for (const ox of oxs) {
   437→            const diff = ox - dx
   438→            if (
   439→              Math.abs(diff) <= threshold &&
   440→              (!bestV || Math.abs(diff) < Math.abs(bestV.diff))
   441→            ) {
   442→              bestV = { diff, line: ox }
   443→            }
   444→          }
   445→        }
   446→        for (const dy of dys) {
   447→          for (const oy of oys) {
   448→            const diff = oy - dy
   449→            if (
   450→              Math.abs(diff) <= threshold &&
   451→              (!bestH || Math.abs(diff) < Math.abs(bestH.diff))
   452→            ) {
   453→              bestH = { diff, line: oy }
   454→            }
   455→          }
   456→        }
   457→      }
   458→
   459→      const snapX = bestV?.diff ?? 0
   460→      const snapY = bestH?.diff ?? 0
   461→      if (snapX !== 0 || snapY !== 0) {
   462→        onNodesChange([
   463→          {
   464→            id: dragged.id,
   465→            type: 'position',
   466→            position: { x: dragged.position.x + snapX, y: dragged.position.y + snapY },
   467→          },
   468→        ])
   469→      }
   470→      useCanvasStore.getState().setGuides({
   471→        vertical: bestV ? [bestV.line] : [],
   472→        horizontal: bestH ? [bestH.line] : [],
   473→      })
   474→    },
   475→    [getZoom, onNodesChange],
   476→  )

   485→        store.setGuides(null)

   487→      }
   488→      applyAlignment(dragged, useCanvasStore.getState().nodes)

   722→                          key={s.type}
   723→                          icon={

   737→            </div>

   739→

   743→              className={cn(

   797→/** 分组右键菜单项：重命名 / 换色 / 解组 / 删除分组与成员 */

   799→  groupId,

   802→  groupId: string
   803→  closeMenu: () => void
   804→}) {

   806→  if (!group) return null

   809→    <>

   815→        icon={<Play className="h-3.5 w-3.5 text-amber-300" />}
   816→        label="运行分组（自动补齐上游）"

   818→          void runGroup(group.id)
   819→          closeMenu()
   820→        }}
   821→      />
   822→      <MenuItem
   823→        icon={
   824→          group.collapsed ? (
   825→            <ChevronsRight className="h-3.5 w-3.5 text-emerald-300" />
   826→          ) : (
   827→            <ChevronsLeft className="h-3.5 w-3.5 text-sky-300" />
   828→          )
   829→        }

   832→          useCanvasStore.getState().toggleGroupCollapse(group.id)

   837→        icon={<FoldVertical className="h-3.5 w-3.5 text-zinc-300" />}

   856→          if (name && name.trim()) {
   857→            useCanvasStore.getState().renameGroup(group.id, name.trim())
   858→          }
   859→          closeMenu()
   860→        }}
   861→      />
   862→      <div className="flex items-center gap-1.5 px-3 py-1.5">
   863→        <span className="text-[10px] text-zinc-500">颜色</span>
   864→        <span className="flex items-center gap-1">
   865→          {GROUP_COLORS.map((c) => (
   866→            <button
   867→              key={c.key}
   868→              onClick={() => useCanvasStore.getState().setGroupColor(group.id, c.key)}

   870→                'h-3.5 w-3.5 rounded-full border transition hover:scale-110',
   871→                c.dot,
